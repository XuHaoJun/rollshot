# P3 Fast NCC Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the two-pass full-resolution NCC refine stage with integral-image normalization plus a `wide::f32x8` cross-term dot product, while preserving matcher outcomes and benchmark-reporting the speedup.

**Architecture:** `PreparedFrame` gets a lazy `IntegralImage` cache built from its existing grayscale buffer. The template refine path passes `&PreparedFrame` into NCC scoring so each offset can use cached integral sums plus a private `wide` dot helper for `sum_xy`. The legacy two-pass NCC stays available only in tests for score-equivalence checks.

**Tech Stack:** Rust 1.89, `rollshot-core`, `wide 1.4.0`, `rayon`, existing `StitchMetrics`, existing `stitch_sequences` benchmark harness.

Add this after `Region`:

```rust
#[derive(Debug)]
struct IntegralImage {
    width: usize,
    height: usize,
    sum: Vec<f64>,
    sum_sq: Vec<f64>,
}

impl IntegralImage {
    fn from_gray_f32(gray: &[f32], width: usize, height: usize) -> Self {
        assert_eq!(
            gray.len(),
            width.saturating_mul(height),
            "gray buffer must match integral dimensions"
        );

        let stride = width + 1;
        let mut sum = vec![0.0; (width + 1) * (height + 1)];
        let mut sum_sq = vec![0.0; (width + 1) * (height + 1)];

        for y in 0..height {
            let mut row_sum = 0.0f64;
            let mut row_sum_sq = 0.0f64;
            for x in 0..width {
                let v = gray[y * width + x] as f64;
                row_sum += v;
                row_sum_sq += v * v;

                let idx = (y + 1) * stride + (x + 1);
                sum[idx] = sum[idx - stride] + row_sum;
                sum_sq[idx] = sum_sq[idx - stride] + row_sum_sq;
            }
        }

        Self {
            width,
            height,
            sum,
            sum_sq,
        }
    }

    fn rect_sum(&self, x: usize, y: usize, w: usize, h: usize) -> f64 {
        self.rect(&self.sum, x, y, w, h)
    }

    fn rect_sum_sq(&self, x: usize, y: usize, w: usize, h: usize) -> f64 {
        self.rect(&self.sum_sq, x, y, w, h)
    }

    fn rect(&self, table: &[f64], x: usize, y: usize, w: usize, h: usize) -> f64 {
        debug_assert!(x + w <= self.width);
        debug_assert!(y + h <= self.height);

        let stride = self.width + 1;
        let x2 = x + w;
        let y2 = y + h;
        table[y2 * stride + x2] - table[y * stride + x2] - table[y2 * stride + x]
            + table[y * stride + x]
    }
}
```

```rust
#[cfg(test)]
fn legacy_ncc_score_shifted(
    prev_gray: &[f32],
    curr_gray: &[f32],
    width: u32,
    height: u32,
    region: Region,
    dx: i32,
    dy: i32,
) -> f32 {
    let overlap = match compute_overlap(width, height, width, height, dx, dy) {
        Some(overlap) => overlap,
        None => return f32::MIN,
    };
    let x0 = region.x.max(overlap.prev_x);
    let y0 = region.y.max(overlap.prev_y);
    let x1 = (region.x + region.w).min(overlap.prev_x + overlap.width);
    let y1 = (region.y + region.h).min(overlap.prev_y + overlap.height);
    if x1 <= x0 || y1 <= y0 {
        return f32::MIN;
    }

    let mut prev_sum = 0.0f32;
    let mut curr_sum = 0.0f32;
    let mut count = 0usize;
    for prev_y in y0..y1 {
        for prev_x in x0..x1 {
            let curr_x = (prev_x as i32 - dx) as u32;
            let curr_y = (prev_y as i32 - dy) as u32;
            let prev_idx = (prev_y * width + prev_x) as usize;
            let curr_idx = (curr_y * width + curr_x) as usize;
            prev_sum += prev_gray[prev_idx];
            curr_sum += curr_gray[curr_idx];
            count += 1;
        }
    }
    if count == 0 {
        return f32::MIN;
    }

    let prev_mean = prev_sum / count as f32;
    let curr_mean = curr_sum / count as f32;
    let mut num = 0.0f32;
    let mut prev_var = 0.0f32;
    let mut curr_var = 0.0f32;
    for prev_y in y0..y1 {
        for prev_x in x0..x1 {
            let curr_x = (prev_x as i32 - dx) as u32;
            let curr_y = (prev_y as i32 - dy) as u32;
            let prev_idx = (prev_y * width + prev_x) as usize;
            let curr_idx = (curr_y * width + curr_x) as usize;
            let p = prev_gray[prev_idx] - prev_mean;
            let c = curr_gray[curr_idx] - curr_mean;
            num += p * c;
            prev_var += p * p;
            curr_var += c * c;
        }
    }

    if prev_var <= 1.0 || curr_var <= 1.0 {
        return f32::MIN;
    }
    num / (prev_var.sqrt() * curr_var.sqrt())
}
```

```rust
fn clipped_ncc_rects(
    width: u32,
    height: u32,
    region: Region,
    dx: i32,
    dy: i32,
) -> Option<(Region, Region)> {
    let overlap = compute_overlap(width, height, width, height, dx, dy)?;
    let x0 = region.x.max(overlap.prev_x);
    let y0 = region.y.max(overlap.prev_y);
    let x1 = (region.x + region.w).min(overlap.prev_x + overlap.width);
    let y1 = (region.y + region.h).min(overlap.prev_y + overlap.height);
    if x1 <= x0 || y1 <= y0 {
        return None;
    }

    let prev_rect = Region {
        x: x0,
        y: y0,
        w: x1 - x0,
        h: y1 - y0,
    };
    let curr_rect = Region {
        x: (x0 as i32 - dx) as u32,
        y: (y0 as i32 - dy) as u32,
        w: prev_rect.w,
        h: prev_rect.h,
    };
    Some((prev_rect, curr_rect))
}

fn dot_wide_f32(
    prev_gray: &[f32],
    curr_gray: &[f32],
    width: usize,
    prev_rect: Region,
    curr_rect: Region,
) -> f64 {
    let mut lanes = f32x8::ZERO;
    let mut tail = 0.0f64;
    let row_width = prev_rect.w as usize;

    for row in 0..prev_rect.h as usize {
        let prev_start = (prev_rect.y as usize + row) * width + prev_rect.x as usize;
        let curr_start = (curr_rect.y as usize + row) * width + curr_rect.x as usize;
        let mut col = 0usize;

        while col + 8 <= row_width {
            let prev_chunk = f32x8::from([
                prev_gray[prev_start + col],
                prev_gray[prev_start + col + 1],
                prev_gray[prev_start + col + 2],
                prev_gray[prev_start + col + 3],
                prev_gray[prev_start + col + 4],
                prev_gray[prev_start + col + 5],
                prev_gray[prev_start + col + 6],
                prev_gray[prev_start + col + 7],
            ]);
            let curr_chunk = f32x8::from([
                curr_gray[curr_start + col],
                curr_gray[curr_start + col + 1],
                curr_gray[curr_start + col + 2],
                curr_gray[curr_start + col + 3],
                curr_gray[curr_start + col + 4],
                curr_gray[curr_start + col + 5],
                curr_gray[curr_start + col + 6],
                curr_gray[curr_start + col + 7],
            ]);
            lanes += prev_chunk * curr_chunk;
            col += 8;
        }

        while col < row_width {
            tail += (prev_gray[prev_start + col] as f64) * (curr_gray[curr_start + col] as f64);
            col += 1;
        }
    }

    lanes.reduce_add() as f64 + tail
}
```

Add this function:

```rust
fn fast_ncc_score_shifted(
    prev: &PreparedFrame,
    curr: &PreparedFrame,
    region: Region,
    dx: i32,
    dy: i32,
) -> f32 {
    let (width, height) = prev.dimensions();
    let (prev_rect, curr_rect) = match clipped_ncc_rects(width, height, region, dx, dy) {
        Some(rects) => rects,
        None => return f32::MIN,
    };

    #[cfg(test)]
    with_active_search_budget(|budget| {
        budget.full_res_ncc_calls += 1;
        budget.full_res_ncc_pixel_visits += u64::from(prev_rect.w) * u64::from(prev_rect.h);
    });

    let n = f64::from(prev_rect.w) * f64::from(prev_rect.h);
    if n == 0.0 {
        return f32::MIN;
    }

    let prev_integral = prev.integral();
    let curr_integral = curr.integral();
    let sum_prev = prev_integral.rect_sum(
        prev_rect.x as usize,
        prev_rect.y as usize,
        prev_rect.w as usize,
        prev_rect.h as usize,
    );
    let sum_prev_sq = prev_integral.rect_sum_sq(
        prev_rect.x as usize,
        prev_rect.y as usize,
        prev_rect.w as usize,
        prev_rect.h as usize,
    );
    let sum_curr = curr_integral.rect_sum(
        curr_rect.x as usize,
        curr_rect.y as usize,
        curr_rect.w as usize,
        curr_rect.h as usize,
    );
    let sum_curr_sq = curr_integral.rect_sum_sq(
        curr_rect.x as usize,
        curr_rect.y as usize,
        curr_rect.w as usize,
        curr_rect.h as usize,
    );
    let sum_xy = dot_wide_f32(
        prev.gray(),
        curr.gray(),
        width as usize,
        prev_rect,
        curr_rect,
    );

    let numerator = sum_xy - (sum_prev * sum_curr / n);
    let prev_var = sum_prev_sq - (sum_prev * sum_prev / n);
    let curr_var = sum_curr_sq - (sum_curr * sum_curr / n);
    if prev_var <= 1e-9 || curr_var <= 1e-9 {
        return f32::MIN;
    }

    (numerator / (prev_var * curr_var).sqrt()) as f32
}
```
